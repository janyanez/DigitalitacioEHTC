// --- FUNCIONS D'AJUDA PER A GEMINI API ---

const fetchWithRetry = async (url, options, retries = 5) => {
  const delays = [1000, 2000, 4000, 8000, 16000];
  for (let i = 0; i < retries; i++) {
    try {
      const response = await fetch(url, options);
      if (!response.ok) {
        const errorData = await response.json().catch(() => ({}));
        throw new Error(`HTTP error! status: ${response.status}, message: ${JSON.stringify(errorData)}`);
      }
      return await response.json();
    } catch (error) {
      if (i === retries - 1) throw error;
      await new Promise(res => setTimeout(res, delays[i]));
    }
  }
};

// Converteix PCM16 a WAV per reproduir l'àudio de Gemini TTS
const pcmToWav = (pcmData, sampleRate = 24000) => {
  const numChannels = 1;
  const bitsPerSample = 16;
  const byteRate = sampleRate * numChannels * (bitsPerSample / 8);
  const blockAlign = numChannels * (bitsPerSample / 8);
  const dataSize = pcmData.length;
  const buffer = new ArrayBuffer(44 + dataSize);
  const view = new DataView(buffer);

  const writeString = (offset, string) => {
    for (let i = 0; i < string.length; i++) {
      view.setUint8(offset + i, string.charCodeAt(i));
    }
  };

  writeString(0, 'RIFF');
  view.setUint32(4, 36 + dataSize, true);
  writeString(8, 'WAVE');
  writeString(12, 'fmt ');
  view.setUint32(16, 16, true);
  view.setUint16(20, 1, true);
  view.setUint16(22, numChannels, true);
  view.setUint32(24, sampleRate, true);
  view.setUint32(28, byteRate, true);
  view.setUint16(32, blockAlign, true);
  view.setUint16(34, bitsPerSample, true);
  writeString(36, 'data');
  view.setUint32(40, dataSize, true);

  const pcmBytes = new Uint8Array(pcmData);
  const wavBytes = new Uint8Array(buffer, 44);
  wavBytes.set(pcmBytes);

  return new Blob([buffer], { type: 'audio/wav' });
};

      setIsChatting(false);

    try {
      const payload = {
        contents: [{ parts: [{ text: `Di-ho amb un to alegre i amable en català: ${text}` }] }],

          responseModalities: ["AUDIO"],

            ><Star size={12} /> Gourmet</button>
            <button 
              onClick={() => setTimePref('batch')}
              className={`flex-1 py-1.5 text-[11px] rounded-lg border font-medium transition-colors flex items-center justify-center gap-1 ${timePref === 'batch' ? 'bg-emerald-100 border-emerald-300 text-emerald-700 shadow-sm' : 'bg-gray-50 border-gray-200 text-gray-600 hover:bg-gray-100'}`}
            ><Layers size={12} /> Batch</button>
          </div>
        </div>
      </div>
      
      <div className="w-full space-y-4 relative">
        <button 
          onClick={() => fileInputRef.current?.click()}
          className="w-full bg-green-600 hover:bg-green-700 text-white font-semibold py-4 px-6 rounded-2xl shadow-xl flex items-center justify-center gap-3 transition-transform active:scale-95"
        >
          <Camera size={24} />
          Fes una foto a la nevera
        </button>
        <button 
          onClick={() => fileInputRef.current?.click()}
          className="w-full bg-white border-2 border-green-100 text-green-700 font-semibold py-4 px-6 rounded-2xl shadow-sm flex items-center justify-center gap-3 transition-transform active:scale-95"
        >
          <ImageIcon size={24} />
          Puja una imatge de la galeria
        </button>
        <input 
          type="file" 
          accept="image/*" 
          ref={fileInputRef} 
          className="hidden" 
          onChange={handleImageUpload} 
        />
      </div>
    </div>
  );

  const renderAnalyzingView = () => (
    <div className="flex flex-col items-center justify-center h-full p-6 text-center bg-green-50">
      <div className="relative mb-8">
        {imageSrc && (
          <img src={imageSrc} alt="Nevera" className="w-48 h-48 object-cover rounded-2xl shadow-2xl opacity-50" />
        )}
        <div className="absolute inset-0 flex items-center justify-center">
          <Loader2 className="w-12 h-12 text-green-600 animate-spin" />
        </div>
      </div>
      <h2 className="text-2xl font-bold text-gray-800 mb-2">
        {timePref !== 'mix' || temperaturePref !== 'qualsevol' ? "Aplicant preferències i repensant..." : "Analitzant la imatge..."}
      </h2>
      <p className="text-gray-600">Comprovant què podem cuinar amb això...</p>
    </div>
  );

  const renderRecipesView = () => (
    <div className="flex flex-col h-full bg-gray-50">
      <div className="p-6 bg-white shadow-sm rounded-b-3xl z-10">
        <button onClick={() => setView('home')} className="mb-4 text-green-600 flex items-center text-sm font-medium">
          <ArrowLeft size={16} className="mr-1"/> Tornar enrere
        </button>
        <h2 className="text-2xl font-bold text-gray-800">Què et ve de gust avui?</h2>
        <div className="mt-3 flex flex-wrap gap-2">
          <span className="text-xs font-semibold bg-green-100 text-green-800 px-3 py-1 rounded-full flex items-center">
            <Leaf size={12} className="mr-1"/> Ingredients detectats:
          </span>
          <p className="text-sm text-gray-600 w-full mt-1">
            {analysisResult?.ingredients_detectats.join(", ")}
          </p>
        </div>

        <div className="mt-4 bg-gray-50 p-3 rounded-xl border border-gray-100">
          <div className="flex justify-between items-center mb-2">
            <label className="text-[10px] font-bold text-gray-500 flex items-center uppercase tracking-wider">
              <Settings2 size={12} className="mr-1"/> Preferències
            </label>
            <button 
              onClick={handleRegenerate}
              className="text-[10px] bg-green-100 text-green-700 px-2 py-1.5 rounded-md hover:bg-green-200 transition-colors flex items-center font-bold"
            >
              <RefreshCw size={10} className="mr-1" /> Aplicar i refer
            </button>
          </div>
          
          <div className="space-y-1.5">
            <div className="flex gap-1.5">
              <button 
                onClick={() => setTemperaturePref('qualsevol')}
                className={`flex-1 py-1 text-[10px] rounded border font-medium transition-colors ${temperaturePref === 'qualsevol' ? 'bg-gray-800 text-white border-gray-800' : 'bg-white text-gray-600 border-gray-200 hover:bg-gray-50'}`}
              >Mix</button>
              <button 
                onClick={() => setTemperaturePref('fred')}
                className={`flex-[1.5] py-1 text-[10px] rounded border font-medium flex items-center justify-center gap-1 transition-colors ${temperaturePref === 'fred' ? 'bg-blue-100 text-blue-700 border-blue-200' : 'bg-white text-gray-600 border-gray-200 hover:bg-gray-50'}`}
              ><Snowflake size={10} /> Fred</button>
              <button 
                onClick={() => setTemperaturePref('calent')}
                className={`flex-[1.5] py-1 text-[10px] rounded border font-medium flex items-center justify-center gap-1 transition-colors ${temperaturePref === 'calent' ? 'bg-orange-100 text-orange-700 border-orange-200' : 'bg-white text-gray-600 border-gray-200 hover:bg-gray-50'}`}
              ><Flame size={10} /> Calent</button>
            </div>
            
            <div className="flex gap-1.5">
              <button 
                onClick={() => setTimePref('mix')}
                className={`flex-1 py-1 text-[10px] rounded border font-medium transition-colors ${timePref === 'mix' ? 'bg-gray-800 text-white border-gray-800' : 'bg-white text-gray-600 border-gray-200 hover:bg-gray-50'}`}
              >Tots</button>
              <button 
                onClick={() => setTimePref('express')}
                className={`flex-[1.2] py-1 text-[10px] rounded border font-medium flex items-center justify-center gap-1 transition-colors ${timePref === 'express' ? 'bg-yellow-100 text-yellow-700 border-yellow-200' : 'bg-white text-gray-600 border-gray-200 hover:bg-gray-50'}`}
              ><Zap size={10} /> Expr.</button>
              <button 
                onClick={() => setTimePref('gourmet')}
                className={`flex-[1.2] py-1 text-[10px] rounded border font-medium flex items-center justify-center gap-1 transition-colors ${timePref === 'gourmet' ? 'bg-purple-100 text-purple-700 border-purple-200' : 'bg-white text-gray-600 border-gray-200 hover:bg-gray-50'}`}
              ><Star size={10} /> Gour.</button>
              <button 
                onClick={() => setTimePref('batch')}
                className={`flex-[1.2] py-1 text-[10px] rounded border font-medium flex items-center justify-center gap-1 transition-colors ${timePref === 'batch' ? 'bg-emerald-100 text-emerald-700 border-emerald-200' : 'bg-white text-gray-600 border-gray-200 hover:bg-gray-50'}`}
              ><Layers size={10} /> Batch</button>
            </div>
          </div>
        </div>
      </div>

      <div className="flex-1 overflow-y-auto p-4 space-y-4">
        {analysisResult?.receptes.map((recipe, index) => {
          let badgeColor = "bg-blue-100 text-blue-800";
          if (recipe.categoria.toLowerCase().includes("gourmet")) badgeColor = "bg-purple-100 text-purple-800";
          if (recipe.categoria.toLowerCase().includes("batch")) badgeColor = "bg-orange-100 text-orange-800";
          
          const imgStatus = recipeImages[recipe.titol];

          return (
            <div 
              key={index} 
              onClick={() => startCooking(recipe)}
              className="bg-white rounded-2xl p-4 shadow-sm border border-gray-100 cursor-pointer hover:shadow-md transition-shadow active:scale-[0.98]"
            >
              <div className="flex justify-between items-start mb-3">
                <span className={`text-[10px] font-bold px-2 py-1 rounded-full uppercase tracking-wider ${badgeColor}`}>
                  {recipe.categoria}
                </span>
                <div className="flex items-center text-gray-500 text-xs font-medium">
                  <Clock size={14} className="mr-1" />
                  {recipe.temps}
                </div>
              </div>
              
              <div className="flex gap-4 items-center">
                <div className="w-20 h-20 rounded-xl bg-gray-100 flex-shrink-0 overflow-hidden relative border border-gray-200">
                  {imgStatus === 'error' ? (
                    <div className="w-full h-full flex items-center justify-center text-gray-400">
                      <ImageOff size={24} />
                    </div>
                  ) : imgStatus ? (
                    <img src={imgStatus} alt={recipe.titol} className="w-full h-full object-cover" />
                  ) : (
                    <div className="w-full h-full flex items-center justify-center">
                      <Loader2 size={20} className="animate-spin text-green-500" />
                    </div>
                  )}
                </div>
                
                <div className="flex-1">
                  <h3 className="text-[17px] font-bold text-gray-900 leading-tight">{recipe.titol}</h3>
                  <p className="text-xs text-gray-500 mt-1.5 line-clamp-2">{recipe.descripcio_breu}</p>
                </div>
              </div>
              
              <div className="mt-4 pt-3 border-t border-gray-50 flex justify-between items-center text-green-600 font-medium text-sm">
                Veure recepta pas a pas <ChevronRight size={18} />
              </div>
            </div>
          );
        })}
      </div>
    </div>
  );

  const renderCookingView = () => {
    const imgStatus = recipeImages[selectedRecipe?.titol];

    return (
      <div className="flex flex-col h-full bg-white relative">
        <div className="p-5 pb-0 shadow-sm z-10 bg-white sticky top-0">
          <div className="flex justify-between items-center mb-3">
            <button onClick={() => setView('recipes')} className="text-gray-500 flex items-center text-sm">
              <ArrowLeft size={16} className="mr-1"/> Tornar a les opcions
            </button>
            <button onClick={handleShare} className="text-green-600 flex items-center text-sm bg-green-50 px-3 py-1 rounded-full font-medium transition-colors hover:bg-green-100">
              <Share2 size={14} className="mr-1.5"/> Compartir
            </button>
          </div>
          
          <div className="w-full h-40 bg-gray-100 rounded-2xl overflow-hidden relative border border-gray-200 mb-4 mt-2">
            {imgStatus === 'error' ? (
              <div className="w-full h-full flex items-center justify-center text-gray-400 gap-2">
                <ImageOff size={28} />
                <span className="text-sm font-medium">Sense imatge</span>
              </div>
            ) : imgStatus ? (
              <img src={imgStatus} alt={selectedRecipe?.titol} className="w-full h-full object-cover" />
            ) : (
              <div className="w-full h-full flex flex-col items-center justify-center text-gray-400 gap-2">
                <Loader2 size={24} className="animate-spin text-green-500" />
                <span className="text-xs font-medium">Generant foto del plat...</span>
              </div>
            )}
          </div>
          
          <h2 className="text-xl font-bold text-gray-900 leading-tight">{selectedRecipe?.titol}</h2>
          <div className="flex items-center gap-4 mt-2 text-sm text-gray-500 pb-4 border-b border-gray-100">
            <span className="flex items-center"><Clock size={14} className="mr-1"/> {selectedRecipe?.temps}</span>
            <span>Dificultat: {selectedRecipe?.dificultat}</span>
          </div>
          
          <div className="py-3 bg-white">
            <div className="bg-orange-50 p-2.5 rounded-xl border border-orange-100 flex gap-2">
              <input 
                type="text"
                value={recipeModifier}
                onChange={(e) => setRecipeModifier(e.target.value)}
                placeholder="Ex: Fes-la vegana, menys sal..."
                className="flex-1 text-sm px-3 py-1.5 rounded-lg border border-orange-200 focus:outline-none focus:ring-2 focus:ring-orange-500 bg-white"
                disabled={isModifyingRecipe}
              />
              <button 
                onClick={handleModifyRecipe}
                disabled={isModifyingRecipe || !recipeModifier.trim()}
                className="bg-orange-100 text-orange-700 px-3 py-1.5 rounded-lg hover:bg-orange-200 transition-colors flex items-center justify-center disabled:opacity-50"
                title="Modificar recepta"
              >
                {isModifyingRecipe ? <Loader2 size={16} className="animate-spin" /> : <Sparkles size={16} />}
              </button>
            </div>
          </div>
        </div>

        <div className="flex-1 overflow-y-auto pb-40">
          <div className="p-6 pt-2">
            <h3 className="font-bold text-gray-800 mb-3 flex items-center">
              <Leaf size={18} className="mr-2 text-green-600"/> Ingredients
            </h3>
            <ul className="space-y-2 mb-8">
              {selectedRecipe?.ingredients_necessaris.map((ing, i) => (
                <li key={i} className="flex items-start text-gray-600 text-sm">
                  <div className="min-w-2 mt-1.5 mr-2 w-2 h-2 rounded-full bg-green-400"></div>
                  {ing}
                </li>
              ))}
            </ul>

            <h3 className="font-bold text-gray-800 mb-3 flex items-center">
              <ChefHat size={18} className="mr-2 text-green-600"/> Passos
            </h3>
            <ol className="space-y-4">
              {selectedRecipe?.passos.map((step, i) => (
                <li key={i} className="flex gap-3 text-sm text-gray-600">
                  <span className="font-bold text-green-600 min-w-4">{i + 1}.</span>
                  <span>{step}</span>
                </li>
              ))}
            </ol>
          </div>

          <div className="px-4 pb-4">
            <div className="flex items-center justify-center my-6">
              <div className="h-px bg-gray-200 flex-1"></div>
              <span className="px-3 text-xs text-gray-400 font-medium uppercase tracking-wider">Assistent de cuina</span>
              <div className="h-px bg-gray-200 flex-1"></div>
            </div>
            
            <div className="space-y-4">
              {chatMessages.map((msg, i) => (
                <div key={i} className={`flex ${msg.role === 'user' ? 'justify-end' : 'justify-start'}`}>
                  <div className={`max-w-[85%] rounded-2xl p-3 text-sm shadow-sm ${
                    msg.role === 'user' 
                      ? 'bg-green-600 text-white rounded-br-none' 
                      : 'bg-gray-100 text-gray-800 rounded-bl-none'
                  }`}>
                    {msg.role === 'assistant' && i === 0 && <ChefHat size={16} className="inline mr-2 -mt-1 text-green-600"/>}
                    <span dangerouslySetInnerHTML={{ __html: msg.text.replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>') }} />
                  </div>
                </div>
              ))}
              {isChatting && (
                <div className="flex justify-start">
                  <div className="bg-gray-100 rounded-2xl p-4 rounded-bl-none flex gap-1">
                    <div className="w-2 h-2 bg-gray-400 rounded-full animate-bounce"></div>
                    <div className="w-2 h-2 bg-gray-400 rounded-full animate-bounce" style={{animationDelay: '0.2s'}}></div>
                    <div className="w-2 h-2 bg-gray-400 rounded-full animate-bounce" style={{animationDelay: '0.4s'}}></div>
                  </div>
                </div>
              )}
              <div ref={chatEndRef} />
            </div>
          </div>
        </div>

        <div className="absolute bottom-0 left-0 right-0 p-4 bg-gradient-to-t from-white via-white to-transparent pt-10">
          <div className="flex items-end gap-2 bg-white border border-gray-200 shadow-lg rounded-3xl p-2 pl-4">
            <textarea 
              value={chatInput}
              onChange={(e) => setChatInput(e.target.value)}
              onKeyDown={handleKeyPress}
              placeholder="Ex: No tinc llimona, què faig?"
              className="flex-1 bg-transparent border-none focus:outline-none text-sm resize-none max-h-24 py-2 text-gray-800"
              rows="1"
            />
            <button 
              onClick={toggleSpeechRecognition}
              className={`p-3 rounded-full flex-shrink-0 transition-colors ${isListening ? 'bg-red-100 text-red-600 animate-pulse' : 'bg-gray-100 text-gray-600 hover:bg-gray-200'}`}
            >
              <Mic size={18} />
            </button>
            <button 
              onClick={sendMessage}
              disabled={!chatInput.trim() || isChatting}
              className={`p-3 rounded-full flex-shrink-0 transition-colors ${chatInput.trim() ? 'bg-green-600 text-white shadow-md' : 'bg-green-100 text-green-400'}`}
            >
              <Send size={18} className="ml-0.5" />
            </button>
          </div>
        </div>
      </div>
    );
  };

  return (
    <div className="w-full min-h-screen bg-gray-900 flex justify-center items-center p-0 md:p-6 font-sans">
      <div className="w-full h-[100dvh] md:h-[850px] md:max-w-[400px] bg-white md:rounded-[2.5rem] md:shadow-2xl overflow-hidden relative ring-1 ring-gray-200">
        <div className="h-6 w-full bg-transparent absolute top-0 z-50 flex justify-center hidden md:flex">
          <div className="w-32 h-6 bg-black rounded-b-3xl"></div>
        </div>

        {renderErrorModal()}

        <div className="h-full pt-0 md:pt-6">
          {view === 'home' && renderHomeView()}
          {view === 'analyzing' && renderAnalyzingView()}
          {view === 'recipes' && renderRecipesView()}
          {view === 'cooking' && renderCookingView()}
        </div>
      </div>
    </div>
  );
}
